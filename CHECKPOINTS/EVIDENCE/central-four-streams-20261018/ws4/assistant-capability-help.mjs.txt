// Match the whole turn: qualifiers and extra lines belong to the report flow.
const questions=new Set([
 "βοηθεια","τι μπορεις να κανεις","τι μπορεις να κανεις για μενα",
 "πως μπορεις να με βοηθησεις","τι μπορω να σε ρωτησω",
 "τι αναφορες μπορεις να βγαλεις"
]);
export function assistantCapabilityHelp(question,tools=[]){
 const raw=String(question??'').trim();
 if(/[\r\n]/.test(raw))return null;
 const q=raw.normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLocaleLowerCase('el-GR').replace(/\s+/g,' ').replace(/[;?!.·]+$/u,'').trim();
 if(!questions.has(q))return null;
 const names=new Set(tools.map(t=>t?.name)),highlights=[];
 if(names.has('sales_by_store'))highlights.push('Πωλήσεις και τζίρος, είδη, κατηγορίες και εφαρμοσμένες προσφορές στο ήδη επιλεγμένο εύρος. Π.χ. «Τι τζίρο κάναμε χθες;».');
 else if(names.has('sales_by_product'))highlights.push('Πωλήσεις συγκεκριμένου είδους στο επιλεγμένο κατάστημα. Π.χ. «Πωλήσεις για το είδος ΝΕΡΟ χθες».');
 if(names.has('stock_by_product'))highlights.push('Τρέχον καταγεγραμμένο απόθεμα συγκεκριμένου είδους. Π.χ. «Τι απόθεμα έχει το είδος ΝΕΡΟ;». Δεν είναι φυσική καταμέτρηση ή ιστορικό απόθεμα.');
 if(names.has('cash_details'))highlights.push('Έλεγχος ταμείων από κλεισμένες βάρδιες. Π.χ. «Έλεγχος ταμείων χθες». Δεν περιλαμβάνονται ανοικτές βάρδιες.');
 if(names.has('consumption_details'))highlights.push('Καταχωρημένη ιδιοκατανάλωση και καταστροφή προϊόντων στο επιλεγμένο κατάστημα. Π.χ. «Ιδιοκατανάλωση χθες».');
 if(names.has('business_records'))highlights.push('Μεταφορές ιδιοκτήτη, πληρωμές/έξοδα, παραστατικά αγορών, προμηθευτές, προσωπικό, βάρδιες ταμείων και καταχωρημένα συμβάντα/τραπεζικές κινήσεις, εφόσον επιτρέπεται η αντίστοιχη ενότητα. Π.χ. «Μεταφορές στον ιδιοκτήτη χθες».');
 return {
  answer:highlights.length?'Μπορώ να σε βοηθήσω με τις παρακάτω συνδεδεμένες αναζητήσεις, ανάλογα με τα δικαιώματα και τις ενεργές ενότητες της πρόσβασής σου. Γράψε τι θέλεις, το είδος όπου χρειάζεται και την περίοδο, π.χ. «σήμερα», «χθες» ή ημερομηνίες από–έως.':'Δεν υπάρχει διαθέσιμη συνδεδεμένη αναφορά στο τρέχον εύρος πρόσβασης.',
  highlights,sources:[],limitations:'Μόνο ανάγνωση στο ήδη επιλεγμένο εξουσιοδοτημένο εύρος. Δεν διάβασα οικονομικά στοιχεία για αυτή τη βοήθεια και δεν αλλάζω δεδομένα ή δικαιώματα. Κάθε αναφορά ελέγχει ξανά την πρόσβαση και την πραγματική πηγή· αν λείπει πληροφορία, ζητώ διευκρίνιση.'
 };
}
